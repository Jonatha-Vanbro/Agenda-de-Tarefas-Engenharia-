const APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbycq3UyMHH_c3TtGbJno6mwAiuI0jqjc8IhFjtyCC1fjNCDVBHVhoLeoLbTwtaWnq9T/exec";

const CATEGORIAS_ORDEM = ["GERAL", "SEGURANÇA", "AGENDA MANUTENÇÃO", "ASSUNTOS EM ESPERA", "ATIVIDADES FINALIZADAS"];
const CATEGORIA_COR = {
  "GERAL": "#0B4F8A",
  "SEGURANÇA": "#B3261E",
  "AGENDA MANUTENÇÃO": "#B7791F",
  "ASSUNTOS EM ESPERA": "#6B5CA5",
  "ATIVIDADES FINALIZADAS": "#7A7F87"
};

let tarefasCache = [];
let colapsadas = new Set(); // categorias colapsadas pelo usuário
const modalTarefa = new bootstrap.Modal(document.getElementById('modalTarefa'));

async function carregarTarefas() {
  document.getElementById('areaConteudo').innerHTML = '<div class="loading-state">Carregando tarefas do backend…</div>';
  try {
    const resp = await fetch(APPS_SCRIPT_URL + "?action=listar");
    const json = await resp.json();
    if (!json.ok) throw new Error(json.error || "Erro desconhecido do backend.");
    tarefasCache = json.data;
    renderizar();
  } catch (err) {
    document.getElementById('areaConteudo').innerHTML =
      '<div class="empty-state">Não foi possível carregar as tarefas.<br>' +
      '<span class="mono" style="font-size:.72rem;">' + escapeHtml(err.message) + '</span><br><br>' +
      'Confira se a URL do Apps Script em <code>APPS_SCRIPT_URL</code> está correta e se a implantação ' +
      'está com acesso "Qualquer pessoa".</div>';
  }
}

function aplicarFiltros(lista) {
  const cat = document.getElementById('filtroCategoria').value;
  const prio = document.getElementById('filtroPrioridade').value;
  const resp = document.getElementById('filtroResponsavel').value.trim().toLowerCase();
  const mostrarFinalizadas = document.getElementById('filtroFinalizadas').checked;

  return lista.filter(t => {
    if (cat && t.categoria !== cat) return false;
    if (prio && t.prioridade !== prio) return false;
    if (resp && !(t.responsavel || '').toLowerCase().includes(resp)) return false;
    if (!mostrarFinalizadas && t.concluida === 'SIM' && cat !== 'ATIVIDADES FINALIZADAS') return false;
    return true;
  });
}

function renderizar() {
  const filtradas = aplicarFiltros(tarefasCache);
  const area = document.getElementById('areaConteudo');

  const total = tarefasCache.length;
  const abertas = tarefasCache.filter(t => t.concluida !== 'SIM').length;
  const atrasadas = tarefasCache.filter(t => (t.situacao || '').startsWith('Atrasado')).length;
  document.getElementById('resumoHeader').textContent =
    total + ' tarefa(s) no total · ' + abertas + ' em aberto · ' + atrasadas + ' atrasada(s)';

  if (filtradas.length === 0) {
    area.innerHTML = '<div class="empty-state">Nenhuma tarefa encontrada com esses filtros.</div>';
    return;
  }

  const porCategoria = {};
  filtradas.forEach(t => {
    const c = t.categoria || 'SEM CATEGORIA';
    if (!porCategoria[c]) porCategoria[c] = [];
    porCategoria[c].push(t);
  });

  const categoriasPresentes = CATEGORIAS_ORDEM.filter(c => porCategoria[c]);
  Object.keys(porCategoria).forEach(c => { if (!categoriasPresentes.includes(c)) categoriasPresentes.push(c); });

  area.innerHTML = categoriasPresentes.map(cat => renderCategoria(cat, porCategoria[cat])).join('');

  // listeners
  area.querySelectorAll('.category-header').forEach(h => {
    h.addEventListener('click', () => {
      const cat = h.dataset.cat;
      if (colapsadas.has(cat)) colapsadas.delete(cat); else colapsadas.add(cat);
      renderizar();
    });
  });
  area.querySelectorAll('.btn-editar').forEach(b => {
    b.addEventListener('click', (e) => { e.stopPropagation(); abrirEdicao(b.dataset.id); });
  });
  area.querySelectorAll('.btn-finalizar').forEach(b => {
    b.addEventListener('click', (e) => { e.stopPropagation(); finalizarTarefa(b.dataset.id); });
  });
}

function renderCategoria(cat, tarefas) {
  const cor = CATEGORIA_COR[cat] || '#0B4F8A';
  const isCollapsed = colapsadas.has(cat);
  return `
    <div class="category-panel">
      <div class="category-header ${isCollapsed ? 'collapsed' : ''}" data-cat="${escapeHtml(cat)}" style="--cat-color:${cor};">
        <div>
          <span class="cat-title">${labelCategoria(cat)}</span>
          <span class="cat-count">${tarefas.length} tarefa(s)</span>
        </div>
        <span class="chevron">▾</span>
      </div>
      <div class="category-body ${isCollapsed ? 'hidden' : ''}">
        ${tarefas.map(renderTarefa).join('')}
      </div>
    </div>`;
}

function labelCategoria(cat) {
  const nomes = {
    "GERAL": "Geral",
    "SEGURANÇA": "Segurança",
    "AGENDA MANUTENÇÃO": "Agenda Manutenção",
    "ASSUNTOS EM ESPERA": "Assuntos em Espera",
    "ATIVIDADES FINALIZADAS": "Atividades Finalizadas"
  };
  return nomes[cat] || cat;
}

function renderTarefa(t) {
  const prioClasse = t.prioridade === 'Alta' ? 'alta' : t.prioridade === 'Média' ? 'media' : t.prioridade === 'Baixa' ? 'baixa' : 'none';
  const sit = t.situacao || '';
  let sitClasse = 'vazio';
  if (sit.startsWith('Atrasado')) sitClasse = 'atrasado';
  else if (sit === 'Concluída') sitClasse = 'concluida';
  else if (sit === 'Encerra hoje') sitClasse = 'hoje';
  else if (sit.startsWith('Faltam')) sitClasse = 'ok';

  return `
    <div class="task-row ${t.concluida === 'SIM' ? 'concluida' : ''}">
      <div class="col-registro registro-data">${escapeHtml(t.registro || '—')}</div>
      <div class="col-prio"><span class="badge-prio ${prioClasse}">${escapeHtml(t.prioridade || '—')}</span></div>
      <div class="col-ativ">
        <div class="atividade-title">${escapeHtml(t.atividade)}</div>
        ${t.status ? `<div class="atividade-status">${escapeHtml(t.status)}</div>` : ''}
      </div>
      <div class="col-resp responsavel-txt">${escapeHtml(t.responsavel || '—')}${t.previsao ? '<br><span class="registro-data">prev: ' + escapeHtml(t.previsao) + '</span>' : ''}</div>
      <div class="col-sit"><span class="situacao-txt ${sitClasse}">${escapeHtml(sit || '—')}</span></div>
      <div class="col-acoes">
        <button class="icon-btn btn-editar" data-id="${t.id}" title="Editar">✎</button>
      </div>
      <div class="col-acoes">
        ${t.concluida !== 'SIM' ? `<button class="icon-btn finalizar btn-finalizar" data-id="${t.id}" title="Finalizar">✓</button>` : ''}
      </div>
    </div>`;
}

function escapeHtml(str) {
  if (str === null || str === undefined) return '';
  return String(str)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#039;');
}

/* ---------------- Modal: nova / editar ---------------- */

function abrirNova() {
  document.getElementById('modalTituloTarefa').textContent = 'Nova tarefa';
  document.getElementById('formTarefa').reset();
  document.getElementById('campoId').value = '';
  document.getElementById('campoRegistro').value = new Date().toISOString().slice(0, 10);
  document.getElementById('erroFormulario').textContent = '';
  modalTarefa.show();
}

function abrirEdicao(id) {
  const t = tarefasCache.find(x => String(x.id) === String(id));
  if (!t) return;
  document.getElementById('modalTituloTarefa').textContent = 'Editar tarefa #' + t.id;
  document.getElementById('campoId').value = t.id;
  document.getElementById('campoCategoria').value = t.categoria;
  document.getElementById('campoAtividade').value = t.atividade || '';
  document.getElementById('campoStatus').value = t.status || '';
  document.getElementById('campoPrioridade').value = t.prioridade || '';
  document.getElementById('campoResponsavel').value = t.responsavel || '';
  document.getElementById('campoRegistro').value = paraInputDate(t.registro);
  document.getElementById('campoPrevisao').value = paraInputDate(t.previsao);
  document.getElementById('erroFormulario').textContent = '';
  modalTarefa.show();
}

function paraInputDate(brDate) {
  if (!brDate || !brDate.includes('/')) return '';
  const [d, m, y] = brDate.split('/');
  return `${y}-${m}-${d}`;
}
function paraDataBR(inputDate) {
  if (!inputDate) return '';
  const [y, m, d] = inputDate.split('-');
  return `${d}/${m}/${y}`;
}

async function chamarBackend(action, body) {
  // Content-Type text/plain evita pré-flight CORS no Apps Script.
  const resp = await fetch(APPS_SCRIPT_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
    body: JSON.stringify({ action, ...body })
  });
  const json = await resp.json();
  if (!json.ok) throw new Error(json.error || 'Erro desconhecido do backend.');
  return json.data;
}

  const id = document.getElementById('campoId').value;
  const atividade = document.getElementById('campoAtividade').value.trim();
  const categoria = document.getElementById('campoCategoria').value;
  const erroEl = document.getElementById('erroFormulario');
  erroEl.textContent = '';

  if (!atividade) { erroEl.textContent = 'Descreva a atividade antes de salvar.'; return; }

  const tarefa = {

    atividade,
    status: document.getElementById('campoStatus').value.trim(),
    prioridade: document.getElementById('campoPrioridade').value,
    responsavel: document.getElementById('campoResponsavel').value.trim(),
    registro: paraDataBR(document.getElementById('campoRegistro').value),
    previsao: paraDataBR(document.getElementById('campoPrevisao').value)
  };

  try {
    if (id) {
      await chamarBackend('editar', { id, tarefa });
    } else {
      await chamarBackend('criar', { tarefa });
    }
    modalTarefa.hide();
    await carregarTarefas();
  } catch (err) {
    erroEl.textContent = err.message;
  }
});

async function finalizarTarefa(id) {
  if (!confirm('Deseja realmente finalizar esta atividade?')) return;
  try {
    await chamarBackend('finalizar', { id });
    await carregarTarefas();
  } catch (err) {
    alert('Não foi possível finalizar: ' + err.message);
  }
}
